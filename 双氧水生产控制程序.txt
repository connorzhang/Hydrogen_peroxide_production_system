// 双氧水生产控制系统程序
// 使用S7-200 SMART ST30 PLC
// 版本：V1.0

// *********************
// I/O分配
// *********************
// 输入：
// I0.0 - 开机按钮
// I0.1 - 关机按钮
// I0.2 - 急停按钮
// I0.3 - 复位按钮
// I0.4 - 脉冲流量计1
// I0.5 - 脉冲流量计2

// 输出：
// Q0.0 - 阳极补水电磁阀 / 阳极泵PWM
// Q0.1 - 阴极补水电磁阀 / 阴极泵PWM
// Q0.2 - 阳极控制电磁阀
// Q0.3 - 阴极控制电磁阀
// Q0.4 - 空压机电磁阀
// Q0.5 - 纯水机电磁阀
// Q0.6 - 直流电源开关控制

// *********************
// 变量定义
// *********************
// 标志位
VB0.0 - 系统运行标志
VB0.1 - 开机流程标志
VB0.2 - 关机流程标志
VB0.3 - 急停标志

// 定时器
T37 - 空压机启动延迟（3秒）
T38 - 阳极电磁阀启动延迟（3秒）
T39 - 阴极泵启动延迟（3秒）
T40 - 阴极电磁阀启动延迟（3秒）
T41 - 直流电源启动延迟（5分钟）
T42 - 关机后空压机延迟（30分钟）

// 计数器
C0 - 开机流程步骤计数器
C1 - 关机流程步骤计数器

// 高速计数器
HC0 - 脉冲流量计1计数
HC1 - 脉冲流量计2计数

// 模拟量输入（通过485通信获取）
VD100 - 阳极液位
VD104 - 阴极液位
VD108 - 高压罐压力
VD112 - 压力传感器1
VD116 - 压力传感器2
VD120 - 压力传感器3

// 485通信地址
VB200 - 设备地址（1-6）
VB201 - 通信命令
VD202 - 通信数据

// *********************
// 主程序
// *********************

NETWORK 1
// 初始化
LD     SM0.1
MOVB   0, VB0.0
MOVB   0, VB0.1
MOVB   0, VB0.2
MOVB   0, VB0.3
MOVW   0, C0
MOVW   0, C1

NETWORK 2
// 开机按钮处理
LD     I0.0
O      VB0.1
AN     I0.1
AN     I0.2
=      VB0.1

NETWORK 3
// 关机按钮处理
LD     I0.1
O      VB0.2
AN     I0.0
AN     I0.2
=      VB0.2

NETWORK 4
// 急停按钮处理
LD     I0.2
=      VB0.3

NETWORK 5
// 复位按钮处理
LD     I0.3
AN     VB0.0
R      VB0.3, 1

NETWORK 6
// 开机流程控制
LD     VB0.1
AN     VB0.2
AN     VB0.3
CALL   SBR_0 // 开机流程子程序

NETWORK 7
// 关机流程控制
LD     VB0.2
AN     VB0.1
AN     VB0.3
CALL   SBR_1 // 关机流程子程序

NETWORK 8
// 急停处理
LD     VB0.3
CALL   SBR_2 // 急停处理子程序

NETWORK 9
// 485通信处理
LD     SM0.0
CALL   SBR_3 // 485通信子程序

NETWORK 10
// PWM控制处理
LD     SM0.0
CALL   SBR_4 // PWM控制子程序

NETWORK 11
// 高速计数处理
LD     SM0.0
CALL   SBR_5 // 高速计数子程序

// *********************
// 子程序：开机流程
// *********************
SUBROUTINE SBR_0: 开机流程
NETWORK 1
// 步骤1：总电源
LD     SM0.0
LPS
MOVW   1, C0
=      Q0.6 // 打开直流电源开关
LRD
MOVW   2, C0
=      Q0.5 // 打开纯水机电磁阀
LRD
MOVW   3, C0
=      Q0.0 // 打开阳极补水电磁阀
=      Q0.1 // 打开阴极补水电磁阀
LRD
// 检查液位
LDB=   VB100, 100 // 假设X=100cm
LDB=   VB104, 100
ALD
R      Q0.0, 1 // 关闭阳极补水电磁阀
R      Q0.1, 1 // 关闭阴极补水电磁阀
MOVW   4, C0
LRD
=      Q0.4 // 打开空压机电磁阀
TON    T37, 30 // 3秒延迟
LD     T37
MOVW   5, C0
=      Q0.2 // 打开阳极控制电磁阀
TON    T38, 30 // 3秒延迟
LD     T38
MOVW   6, C0
// 启动阳极泵PWM控制
// 这里需要调用PWM控制子程序
LRD
// 检查压力
LDD>=  VD108, 100000 // 100kPa
ALD
MOVW   7, C0
=      Q0.3 // 打开阴极控制电磁阀
TON    T39, 30 // 3秒延迟
LD     T39
MOVW   8, C0
// 启动阴极泵PWM控制
// 这里需要调用PWM控制子程序
TON    T40, 30 // 3秒延迟
LD     T40
MOVW   9, C0
TON    T41, 30000 // 5分钟延迟
LD     T41
MOVW   10, C0
// 通过485通信控制直流电源输出50A
// 这里需要调用485通信子程序
MOVB   1, VB0.0 // 系统运行标志置1
END_SUBROUTINE

// *********************
// 子程序：关机流程
// *********************
SUBROUTINE SBR_1: 关机流程
NETWORK 1
// 步骤1：关闭直流电源
LD     SM0.0
LPS
MOVW   1, C1
// 通过485通信关闭直流电源
// 这里需要调用485通信子程序
LRD
MOVW   2, C1
R      Q0.0, 1 // 关闭阳极补水电磁阀
R      Q0.1, 1 // 关闭阴极补水电磁阀
LRD
MOVW   3, C1
R      Q0.5, 1 // 关闭纯水机电磁阀
LRD
MOVW   4, C1
R      Q0.3, 1 // 关闭阴极控制电磁阀
// 停止阴极泵PWM控制
LRD
MOVW   5, C1
R      Q0.2, 1 // 关闭阳极控制电磁阀
// 停止阳极泵PWM控制
LRD
MOVW   6, C1
TON    T42, 1800000 // 30分钟延迟
LD     T42
MOVW   7, C1
R      Q0.4, 1 // 关闭空压机电磁阀
LRD
MOVW   8, C1
R      Q0.6, 1 // 关闭直流电源开关
MOVB   0, VB0.0 // 系统运行标志置0
END_SUBROUTINE

// *********************
// 子程序：急停处理
// *********************
SUBROUTINE SBR_2: 急停处理
NETWORK 1
// 关闭所有输出
LD     SM0.0
R      Q0.0, 1
R      Q0.1, 1
R      Q0.2, 1
R      Q0.3, 1
R      Q0.4, 1
R      Q0.5, 1
R      Q0.6, 1
// 停止所有PWM控制
// 关闭直流电源
// 这里需要调用485通信子程序
MOVB   0, VB0.0
MOVB   0, VB0.1
MOVB   0, VB0.2
END_SUBROUTINE

// *********************
// 子程序：485通信
// *********************
SUBROUTINE SBR_3: 485通信
NETWORK 1
// 初始化通信参数
LD     SM0.1
MOVB   96, SMB30 // 9600bps, 8位数据, 1位停止位, 无校验

NETWORK 2
// 轮询设备
LD     SM0.0
LPS
// 读取阳极液位
MOVB   2, VB200
MOVB   3, VB201 // 读取命令
CALL   Modbus_Read
LRD
// 读取阴极液位
MOVB   3, VB200
MOVB   3, VB201
CALL   Modbus_Read
LRD
// 读取高压罐压力
MOVB   4, VB200
MOVB   3, VB201
CALL   Modbus_Read
LRD
// 读取压力传感器1
MOVB   4, VB200
MOVB   3, VB201
CALL   Modbus_Read
LRD
// 读取压力传感器2
MOVB   5, VB200
MOVB   3, VB201
CALL   Modbus_Read
LRD
// 读取压力传感器3
MOVB   6, VB200
MOVB   3, VB201
CALL   Modbus_Read
LRD
// 控制直流电源
LD     VB0.0
MOVB   1, VB200
MOVB   6, VB201 // 写入命令
MOVW   5000, VW202 // 50A
CALL   Modbus_Write
END_SUBROUTINE

// *********************
// 子程序：PWM控制
// *********************
SUBROUTINE SBR_4: PWM控制
NETWORK 1
// 初始化PWM
LD     SM0.1
// 配置Q0.0为PWM输出
MOVB   16#80, SMB67
MOVW   1000, SMW68 // 周期1ms
MOVW   500, SMW70 // 占空比50%
PWM    0, 1

// 配置Q0.1为PWM输出
MOVB   16#80, SMB77
MOVW   1000, SMW78 // 周期1ms
MOVW   500, SMW80 // 占空比50%
PWM    1, 1

NETWORK 2
// 阳极泵PWM控制
LD     VB0.0
LPS
// 根据压力调整占空比
LDD>=  VD108, 100000
MOVW   300, SMW70 // 降低占空比
LRD
LDD<   VD108, 95000
MOVW   700, SMW70 // 增加占空比
END_SUBROUTINE

// *********************
// 子程序：高速计数
// *********************
SUBROUTINE SBR_5: 高速计数
NETWORK 1
// 初始化高速计数器0（流量计1）
LD     SM0.1
MOVB   16#FC, SMB37 // 启用计数
MOVW   0, SMW38 // 当前值
MOVW   0, SMW42 // 预设值
HDEF   0, 0 // 模式0
HSC    0

// 初始化高速计数器1（流量计2）
MOVB   16#FC, SMB47
MOVW   0, SMW48
MOVW   0, SMW52
HDEF   1, 0
HSC    1

NETWORK 2
// 读取计数值
LD     SM0.0
MOVW   HC0, VW300 // 流量计1读数
MOVW   HC1, VW302 // 流量计2读数
END_SUBROUTINE

// *********************
// 中断程序
// *********************
INTERRUPT INT_0: 高速计数中断
NETWORK 1
// 高速计数中断处理
LD     SM0.0
// 处理计数溢出等情况
END_INTERRUPT
